      - item: Tin Ore
        quantity: 80

        quantity: 10

    from: 125
    to: 150

        quantity: 60

    from: 150
    to: 175
    action: >-

        quantity: 60

        quantity: 10

    from: 175
    to: 230
    action: >-

      - item: Mithril Ore
        quantity: 100

  - id: prof-mining-007
    from: 250
    to: 300
    action: >-

    materials:
      - item: Thorium Ore
        quantity: 150
  - id: prof-mining-008
    from: 300
    to: 300
